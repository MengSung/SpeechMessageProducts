let currentLesson = 0;
const totalLessons = 6;
const completed = new Set();

function goTo(n) {
  document.getElementById('lesson' + currentLesson).classList.remove('active');
  document.getElementById('nav' + currentLesson).classList.remove('active');
  completed.add(currentLesson);
  currentLesson = n;
  document.getElementById('lesson' + n).classList.add('active');
  document.getElementById('nav' + n).classList.add('active');
  completed.forEach(i => {
    const nav = document.getElementById('nav' + i);
    if (i !== n) nav.classList.add('done');
  });
  updateProgress();
  window.scrollTo(0,0);
}

function updateProgress() {
  const pct = Math.round((completed.size / (totalLessons + 1)) * 100);
  document.getElementById('progressFill').style.width = pct + '%';
  document.getElementById('progressLabel').textContent = completed.size + ' / ' + (totalLessons + 1) + ' 完成';
}

function answer(el, correct) {
  const opts = el.parentElement.querySelectorAll('.quiz-opt');
  opts.forEach(o => o.style.pointerEvents = 'none');
  const fb = document.getElementById('quizFeedback0');
  if (correct) {
    el.classList.add('correct');
    fb.className = 'quiz-feedback correct';
    fb.textContent = '✓ 正確！第二個提示包含了具體的格式（300字）、語言（繁體中文）、媒介（部落格）、主題和語氣，AI 知道完全知道你要什麼。';

async function callClaude(messages, systemPrompt) {
  const body = {
    model: "claude-sonnet-4-20250514",
    max_tokens: 1000,
    messages: messages
  };
  if (systemPrompt) body.system = systemPrompt;
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body)
  });
  const data = await res.json();
  return data.content?.[0]?.text || "（無回應）";
}

async function runPrompt(inputId, responseId) {
  const prompt = document.getElementById(inputId).value.trim();
  if (!prompt) return;
  const box = document.getElementById(responseId);
  box.className = 'response-box loading';
  box.innerHTML = '<div class="response-label">Claude 的回應</div><div class="response-text placeholder">思考中...</div>';
  try {
    const text = await callClaude([{ role: "user", content: prompt }]);
    box.className = 'response-box';
    box.innerHTML = '<div class="response-label">Claude 的回應</div><div class="response-text">' + escHtml(text) + '</div>';
  } catch(e) {
    box.className = 'response-box';
    box.innerHTML = '<div class="response-label">錯誤</div><div class="response-text placeholder">無法取得回應，請稍後再試。</div>';
  }
}

async function runPromptWithSystem(inputId, systemId, responseId) {
  const prompt = document.getElementById(inputId).value.trim();
  const system = document.getElementById(systemId).value.trim();
  if (!prompt) return;
  const box = document.getElementById(responseId);
  box.className = 'response-box loading';
  box.innerHTML = '<div class="response-label">Claude 的回應</div><div class="response-text placeholder">思考中...</div>';

    const text = await callClaude([{ role: "user", content: prompt }], system || undefined);
    box.className = 'response-box';

    box.innerHTML = '<div class="response-label">錯誤</div><div class="response-text placeholder">無法取得回應，請稍後再試。</div>';

}

  return t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
