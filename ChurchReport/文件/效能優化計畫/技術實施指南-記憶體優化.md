## 一、ToolUtilityClass Dispose Pattern 完整實現

### 問題分析
**當前程式碼問題**:
```csharp
// ? 不完整的 Dispose 實現
protected virtual void Dispose(bool disposing)
{
    if (_disposed) return;
    // 只有註解，沒有實際釋放資源
    // this.m_OrganizationService.Dispose();
    _disposed = true;
}
```

**風險**:
- IOrganizationService 連接未釋放 → 連接洩漏
- FileStream、StreamWriter 未關閉 → 檔案鎖定、記憶體洩漏
- Trace Listener 未移除 → 記憶體累積

### 完整解決方案

#### 1.1 完整的 Dispose 實現
```csharp
// ? 完整的 Dispose Pattern
protected virtual void Dispose(bool disposing)
{
    if (_disposed) return;

    if (disposing)
    {
        // 釋放 Managed 資源
        
        // 1. 釋放 Facade
        _facade?.Dispose();
        
        // 2. 釋放 CRM 連接服務
        (_crmConnectionService as IDisposable)?.Dispose();
        
        // 3. 釋放 Organization Service
        (m_Crm2011OrganizationService as IDisposable)?.Dispose();
        (m_OrganizationService as IDisposable)?.Dispose();
        
        // 4. 釋放追蹤監聽器
        if (m_Listener != null)
        {
            try
            {
#if NET462 || NETFRAMEWORK
                Debug.Listeners.Remove(m_Listener);
#else
                Trace.Listeners.Remove(m_Listener);
#endif
                m_Listener.Flush();
                m_Listener.Close();
                m_Listener.Dispose();
            }
            catch (ObjectDisposedException)
            {
                // 已被釋放，忽略
            }
            finally
            {
                m_Listener = null;
            }
        }
        
        // 5. 釋放檔案寫入器
        if (m_XmlFileStreamWriter != null)
        {
            try
            {
                m_XmlFileStreamWriter.Flush();
                m_XmlFileStreamWriter.Close();
                m_XmlFileStreamWriter.Dispose();
            }
            catch (ObjectDisposedException)
            {
                // 已被釋放，忽略
            }
            finally
            {
                m_XmlFileStreamWriter = null;
            }
        }
        
        // 6. 釋放檔案串流
        if (m_XmlFileStream != null)
        {
            try
            {
                m_XmlFileStream.Flush();
                m_XmlFileStream.Close();
                m_XmlFileStream.Dispose();
            }
            catch (ObjectDisposedException)
            {
                // 已被釋放，忽略
            }
            finally
            {
                m_XmlFileStream = null;
            }
        }
    }

    // 釋放 Unmanaged 資源（如果有）
    // ...

    _disposed = true;
}

public void Dispose()
{
    Dispose(true);
    GC.SuppressFinalize(this);
}

~ToolUtilityClass()
{
    Dispose(false);
}
```

#### 連接池實現
```csharp
// 檔案: ToolUtility\ConnectionOperations\CrmConnectionPool.cs
using System;
using System.Collections.Concurrent;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Xrm.Sdk;

namespace ToolUtilityNameSpace.ConnectionOperations
{
    /// <summary>
    /// CRM 連接池 - 實現 Object Pool Pattern
    /// 遵循 LINUS 原則: 簡潔、高效、可靠
    /// </summary>
    public class CrmConnectionPool : ICrmConnectionPool
    {
        private readonly ConcurrentBag<PooledConnection> _connections;
        private readonly SemaphoreSlim _semaphore;
        private readonly ICrmConnectionService _connectionService;
        private readonly string _serverUrl;
        private readonly string _username;
        private readonly string _password;
        private readonly int _minPoolSize;
        private readonly int _maxPoolSize;
        private readonly TimeSpan _connectionTimeout;
        private readonly TimeSpan _idleTimeout;
        private int _currentSize;
        private bool _disposed;
        private readonly Timer _cleanupTimer;

        public CrmConnectionPool(
            ICrmConnectionService connectionService,
            string serverUrl,
            string username,
            string password,
            int minPoolSize = 3,
            int maxPoolSize = 10,
            TimeSpan? connectionTimeout = null,
            TimeSpan? idleTimeout = null)
        {
            _connectionService = connectionService ?? throw new ArgumentNullException(nameof(connectionService));
            _serverUrl = serverUrl ?? throw new ArgumentNullException(nameof(serverUrl));
            _username = username ?? throw new ArgumentNullException(nameof(username));
            _password = password ?? throw new ArgumentNullException(nameof(password));
            _minPoolSize = minPoolSize;
            _maxPoolSize = maxPoolSize;
            _connectionTimeout = connectionTimeout ?? TimeSpan.FromSeconds(30);
            _idleTimeout = idleTimeout ?? TimeSpan.FromMinutes(10);

            _connections = new ConcurrentBag<PooledConnection>();
            _semaphore = new SemaphoreSlim(maxPoolSize, maxPoolSize);
            _currentSize = 0;

            // 預先創建最小連接數
            InitializeMinConnections();

            // 啟動清理計時器（每分鐘檢查一次）
            _cleanupTimer = new Timer(CleanupIdleConnections, null, TimeSpan.FromMinutes(1), TimeSpan.FromMinutes(1));
        }

        private void InitializeMinConnections()
        {
            for (int i = 0; i < _minPoolSize; i++)
            {
                var connection = CreateConnection();
                _connections.Add(connection);
            }
        }

        private PooledConnection CreateConnection()
        {
            var service = _connectionService.CreateOnPremiseClient(_serverUrl, _username, _password);
            Interlocked.Increment(ref _currentSize);
            
            return new PooledConnection
            {
                Service = service,
                CreatedAt = DateTime.UtcNow,
                LastUsedAt = DateTime.UtcNow,
                IsInUse = false
            };
        }

        public IOrganizationService AcquireConnection()
        {
            if (_disposed)
                throw new ObjectDisposedException(nameof(CrmConnectionPool));

            // 等待可用連接（帶超時）
            if (!_semaphore.Wait(_connectionTimeout))
                throw new TimeoutException("無法在指定時間內取得 CRM 連接");

            try
            {
                // 嘗試從池中取得可用連接
                if (_connections.TryTake(out var pooledConnection))
                {
                    // 檢查連接健康狀態
                    if (IsConnectionHealthy(pooledConnection))
                    {
                        pooledConnection.IsInUse = true;
                        pooledConnection.LastUsedAt = DateTime.UtcNow;
                        return pooledConnection.Service;
                    }
                    else
                    {
                        // 連接不健康，釋放並創建新連接
                        DisposeConnection(pooledConnection);
                        pooledConnection = CreateConnection();
                        pooledConnection.IsInUse = true;
                        pooledConnection.LastUsedAt = DateTime.UtcNow;
                        return pooledConnection.Service;
                    }
                }
                else if (_currentSize < _maxPoolSize)
                {
                    // 池中無可用連接，但未達上限，創建新連接
                    var pooledConnection = CreateConnection();
                    pooledConnection.IsInUse = true;
                    pooledConnection.LastUsedAt = DateTime.UtcNow;
                    return pooledConnection.Service;
                }
                else
                {
                    // 不應該到達這裡
                    throw new InvalidOperationException("連接池邏輯錯誤");
                }
            }
            catch
            {
                _semaphore.Release();
                throw;
            }
        }

        public void ReleaseConnection(IOrganizationService service)
        {
            if (_disposed)
                return;

            if (service == null)
                throw new ArgumentNullException(nameof(service));

            // 找到對應的 PooledConnection
            var pooledConnection = new PooledConnection
            {
                Service = service,
                IsInUse = false,
                LastUsedAt = DateTime.UtcNow
            };

            _connections.Add(pooledConnection);
            _semaphore.Release();
        }

        private bool IsConnectionHealthy(PooledConnection connection)
        {
            try
            {
                // 執行簡單查詢測試連接
                var query = new Microsoft.Xrm.Sdk.Query.QueryExpression("systemuser")
                {
                    ColumnSet = new Microsoft.Xrm.Sdk.Query.ColumnSet("systemuserid"),
                    TopCount = 1
                };
                connection.Service.RetrieveMultiple(query);
                return true;
            }
            catch
            {
                return false;
            }
        }

        private void CleanupIdleConnections(object state)
        {
            if (_disposed)
                return;

            var now = DateTime.UtcNow;
            var connectionsToRemove = new ConcurrentBag<PooledConnection>();

            // 收集需要清理的連接
            while (_connections.TryTake(out var connection))
            {
                if (!connection.IsInUse && (now - connection.LastUsedAt) > _idleTimeout && _currentSize > _minPoolSize)
                {
                    connectionsToRemove.Add(connection);
                }
                else
                {
                    // 保留連接
                    _connections.Add(connection);
                }
            }

            // 釋放閒置連接
            foreach (var connection in connectionsToRemove)
            {
                DisposeConnection(connection);
            }
        }

        private void DisposeConnection(PooledConnection connection)
        {
            try
            {
                (connection.Service as IDisposable)?.Dispose();
                Interlocked.Decrement(ref _currentSize);
            }
            catch
            {
                // 忽略釋放錯誤
            }
        }

        public ConnectionPoolStats GetStats()
        {
            var activeCount = 0;
            var idleCount = 0;

            foreach (var conn in _connections)
            {
                if (conn.IsInUse)
                    activeCount++;
                else
                    idleCount++;
            }

            return new ConnectionPoolStats
            {
                TotalConnections = _currentSize,
                ActiveConnections = activeCount,
                IdleConnections = idleCount,
                WaitingRequests = _maxPoolSize - _semaphore.CurrentCount
            };
        }

        public void Dispose()
        {
            if (_disposed)
                return;

            _disposed = true;
            _cleanupTimer?.Dispose();
            _semaphore?.Dispose();

            // 釋放所有連接
            while (_connections.TryTake(out var connection))
            {
                DisposeConnection(connection);
            }
        }

        private class PooledConnection
        {
            public IOrganizationService Service { get; set; }
            public DateTime CreatedAt { get; set; }
            public DateTime LastUsedAt { get; set; }
            public bool IsInUse { get; set; }
        }
    }
}
```

```csharp
        IHttpContextAccessor httpContextAccessor,

        }
```

### 4.1 記憶體監控中介軟體
```csharp
// 檔案: ChurchReport\Middleware\MemoryMonitoringMiddleware.cs
using Microsoft.AspNetCore.Http;
using System;
using System.Diagnostics;
using System.Threading.Tasks;

namespace ChurchReport.Middleware
{
    /// <summary>
    /// 記憶體監控中介軟體
    /// 遵循 LINUS 原則: 簡潔監控，不影響效能
    /// </summary>
    public class MemoryMonitoringMiddleware
    {
        private readonly RequestDelegate _next;
        private readonly long _warningThreshold = 500 * 1024 * 1024; // 500 MB
        
        public MemoryMonitoringMiddleware(RequestDelegate next)
        {
            _next = next;
        }
        
        public async Task InvokeAsync(HttpContext context)
        {
            var startMemory = GC.GetTotalMemory(false);
            var sw = Stopwatch.StartNew();
            
            try
            {
                await _next(context);
            }
            finally
            {
                sw.Stop();
                var endMemory = GC.GetTotalMemory(false);
                var memoryDelta = endMemory - startMemory;
                
                // 如果單次請求記憶體增長超過閾值，記錄警告
                if (memoryDelta > _warningThreshold)
                {
                    Trace.WriteLine($"[MEMORY WARNING] Path: {context.Request.Path}, " +
                                   $"Memory Delta: {memoryDelta / 1024 / 1024} MB, " +
                                   $"Duration: {sw.ElapsedMilliseconds} ms");
                }
            }
        }
    }
}

// 在 Startup.Configure 中註冊
public void Configure(IApplicationBuilder app, IWebHostEnvironment env)
{
    app.UseMiddleware<MemoryMonitoringMiddleware>();
    // ... 其他中介軟體
}
```

### 4.2 自動記憶體壓力回收
```csharp
// 檔案: ChurchReport\Services\MemoryPressureManager.cs
using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.Extensions.Hosting;

namespace ChurchReport.Services
{
    /// <summary>
    /// 記憶體壓力管理服務
    /// 當記憶體使用超過閾值時，主動執行 GC
    /// </summary>
    public class MemoryPressureManager : IHostedService, IDisposable
    {
        private Timer _timer;
        private readonly long _criticalThreshold = 800 * 1024 * 1024; // 800 MB
        private readonly long _warningThreshold = 600 * 1024 * 1024;  // 600 MB
        
        public Task StartAsync(CancellationToken cancellationToken)
        {
            // 每 30 秒檢查一次記憶體使用
            _timer = new Timer(CheckMemoryPressure, null, TimeSpan.Zero, TimeSpan.FromSeconds(30));
            return Task.CompletedTask;
        }
        
        private void CheckMemoryPressure(object state)
        {
            var currentMemory = GC.GetTotalMemory(false);
            
            if (currentMemory > _criticalThreshold)
            {
                // 嚴重警告: 執行完整 GC
                Trace.WriteLine($"[MEMORY CRITICAL] Current: {currentMemory / 1024 / 1024} MB - Forcing Gen2 GC");
                GC.Collect(2, GCCollectionMode.Aggressive, true, true);
                GC.WaitForPendingFinalizers();
                GC.Collect();
            }
            else if (currentMemory > _warningThreshold)
            {
                // 警告: 執行一般 GC
                Trace.WriteLine($"[MEMORY WARNING] Current: {currentMemory / 1024 / 1024} MB - Forcing Gen1 GC");
                GC.Collect(1, GCCollectionMode.Optimized, false);
            }
        }
        
        public Task StopAsync(CancellationToken cancellationToken)
        {
            _timer?.Change(Timeout.Infinite, 0);
            return Task.CompletedTask;
        }
        
        public void Dispose()
        {
            _timer?.Dispose();
        }
    }
}

// 在 Startup.ConfigureServices 中註冊
public void ConfigureServices(IServiceCollection services)
{
    services.AddHostedService<MemoryPressureManager>();
    // ... 其他服務
}
```
