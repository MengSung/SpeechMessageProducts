## ?? 範例：完整的類別註解

```csharp
using System;

namespace Line.Messaging
{
    /// <summary>
    /// 表示 LINE 音訊訊息
    /// Represents a LINE audio message
    /// </summary>
    /// <remarks>
    /// 音訊訊息用於傳送音訊檔案給使用者。
    /// <para>
    /// Audio message is used to send audio files to users.
    /// </para>
    /// <para>
    /// 音訊規格限制：
    /// - 格式：M4A
    /// - 最大長度：1 分鐘
    /// - 最大檔案：200MB
    /// - HTTPS 協定
    /// </para>
    /// <para>
    /// 官方文件：https://developers.line.biz/en/reference/messaging-api/#audio-message
    /// </para>
    /// </remarks>
    /// <example>
    /// <code>
    /// var message = new AudioMessage(
    ///     "https://example.com/audio.m4a",
    ///     60000
    /// );
    /// await client.ReplyMessageAsync(replyToken, message);
    /// </code>
    /// </example>
    public class AudioMessage : ISendMessage
    {
        /// <summary>
        /// 取得訊息類型，固定為 Audio
        /// Gets the message type, always Audio
        /// </summary>
        public MessageType Type { get; } = MessageType.Audio;

        /// <summary>
        /// 取得或設定快速回覆選單
        /// Gets or sets the quick reply menu
        /// </summary>
        public QuickReply QuickReply { get; set; }

        /// <summary>
        /// 取得音訊檔案的 URL
        /// Gets the URL of the audio file
        /// </summary>
        /// <value>
        /// 音訊檔案 URL（HTTPS，M4A 格式，最大 200MB）
        /// Audio file URL (HTTPS, M4A format, max 200MB)
        /// </value>
        public string OriginalContentUrl { get; }

        /// <summary>
        /// 取得音訊長度（毫秒）
        /// Gets the audio duration in milliseconds
        /// </summary>
        /// <value>
        /// 音訊長度，單位為毫秒（最長 60000ms = 1 分鐘）
        /// Duration in milliseconds (max 60000ms = 1 minute)
        /// </value>
        public long Duration { get; }

        /// <summary>
        /// 初始化 AudioMessage 的新執行個體
        /// Initializes a new instance of the AudioMessage class
        /// </summary>
        /// <param name="originalContentUrl">
        /// 音訊檔案的 HTTPS URL（M4A 格式，最大 200MB）
        /// HTTPS URL of audio file (M4A format, max 200MB)
        /// </param>
        /// <param name="duration">
        /// 音訊長度（毫秒），最長 60000（1 分鐘）
        /// Duration in milliseconds, max 60000 (1 minute)
        /// </param>
        /// <param name="quickReply">
        /// 快速回覆選單（選用）
        /// Quick reply menu (optional)
        /// </param>
        public AudioMessage(string originalContentUrl, long duration, QuickReply quickReply = null)
        {
            OriginalContentUrl = originalContentUrl;
            Duration = duration;
            QuickReply = quickReply;
        }
    }
}
```
