### 第三步: 修復 switch expression

```csharp

private string GetEntityTypeFromId(string id)
{
    return id switch
    {
        "區牧" => "race_id",
        "牧區" => "new_pastoralarea",
        "小組" => "new_smallgroup",
        _ => null
    };
}

// 修改後
private string GetEntityTypeFromId(string id)
{
    switch (id)
    {
        case "區牧":
            return "race_id";
        case "牧區":
            return "new_pastoralarea";
        case "小組":
            return "new_smallgroup";
        default:
            return null;
    }
}
```
